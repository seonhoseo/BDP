# 9. NiFi Access Token 발급
# ==============================================================================

def get_nifi_token():
    """
    NiFi REST API를 사용하기 위한 인증 Token을 발급받습니다.

    --------------------------------------------------------------------------
    왜 Token이 필요한가?
    --------------------------------------------------------------------------

    NiFi는 REST API를 호출할 때 인증이 필요한 경우가 많습니다.

    현재 구성에서는

        NiFi 사용자명
              +
        NiFi 비밀번호
              ↓
        /nifi-api/access/token
              ↓
        JWT Token
              ↓
        NiFi REST API 호출

    순서로 인증합니다.

    --------------------------------------------------------------------------
    처리 순서
    --------------------------------------------------------------------------

    1. NiFi username / password를 준비합니다.

    2. NiFi의 /access/token API를 호출합니다.

    3. NiFi가 인증에 성공하면 Token을 반환합니다.

    4. 반환된 Token을 이후 API 요청에서

        Authorization: Bearer <Token>

    형식으로 사용합니다.

    --------------------------------------------------------------------------
    반환값
    --------------------------------------------------------------------------

    str
        NiFi REST API 호출에 사용할 Access Token
    """

    # --------------------------------------------------------------------------
    # NiFi의 Access Token 발급 API 호출
    # --------------------------------------------------------------------------

    response = requests.post(

        # NiFi REST API의 Token 발급 주소
        f"{NIFI_URL}/nifi-api/access/token",

        # 요청 데이터의 형식
        headers={
            "Content-Type": "application/x-www-form-urlencoded"
        },

        # NiFi 로그인 정보
        data={
            "username": NIFI_USER,
            "password": NIFI_PASSWORD
        },

        # 현재 환경에서는 HTTPS 인증서 검증을 하지 않음
        #
        # 운영환경에서는 정상적인 인증서를 구성하고
        # verify=True 사용을 권장합니다.
        verify=False
    )

    # HTTP 응답이 실패한 경우 예외를 발생시킵니다.
    #
    # 예:
    # 401 → 인증 실패
    # 403 → 권한 부족
    # 500 → NiFi 서버 오류
    # --------------------------------------------------------------------------

    response.raise_for_status()

    # NiFi가 반환한 Token을 문자열로 반환합니다.
    return response.text

# 10. NiFi Process Group 실행 / 중지
# ==============================================================================

def change_pg_state(pg_id, state):
    """
    지정한 NiFi Process Group의 실행 상태를 변경합니다.

    --------------------------------------------------------------------------
    state 값
    --------------------------------------------------------------------------

    RUNNING
        → Process Group 실행

    STOPPED
        → Process Group 중지

    --------------------------------------------------------------------------
    왜 Revision을 먼저 조회하는가?
    --------------------------------------------------------------------------

    NiFi는 여러 사용자가 동시에 Process Group을 수정하는 상황을 방지하기
    위해 Revision 정보를 사용합니다.

    따라서 Process Group 상태를 변경하기 전에
    현재 Revision 정보를 먼저 조회해야 합니다.

    Revision이 맞지 않으면 NiFi가

        HTTP 409 Conflict

    오류를 반환할 수 있습니다.

    --------------------------------------------------------------------------
    처리 순서
    --------------------------------------------------------------------------

    ① NiFi Access Token 발급

    ② Process Group의 현재 정보 조회

    ③ 현재 Revision 정보 추출

    ④ 실행 상태 변경 요청

    --------------------------------------------------------------------------
    Parameters
    --------------------------------------------------------------------------

    pg_id
        변경할 NiFi Process Group ID

    state
        RUNNING 또는 STOPPED
    """

    # --------------------------------------------------------------------------
    # NiFi API 인증을 위한 Token 발급
    # --------------------------------------------------------------------------

    token = get_nifi_token()


    # --------------------------------------------------------------------------
    # REST API 요청 Header 생성
    # --------------------------------------------------------------------------
    #
    # Authorization:
    #     NiFi에게 "나는 인증된 사용자다"라고 알려주는 부분입니다.
    #
    # Bearer:
    #     뒤에 오는 문자열이 Access Token이라는 의미입니다.
    # --------------------------------------------------------------------------

    headers = {

        "Authorization": f"Bearer {token}",

        "Content-Type": "application/json"

    }


    # ==========================================================================
    # 10-1. Process Group 현재 정보 조회
    # ==========================================================================

    response = requests.get(

        # Process Group 정보를 조회하는 NiFi REST API
        f"{NIFI_URL}/nifi-api/process-groups/{pg_id}",

        headers=headers,

        verify=False

    )


    # HTTP 오류가 발생했다면 즉시 예외 발생
    response.raise_for_status()


    # --------------------------------------------------------------------------
    # NiFi가 반환한 JSON에서 revision 정보만 추출합니다.
    #
    # 이 Revision 정보가 있어야 다음 PUT 요청을 통해
    # Process Group의 상태를 변경할 수 있습니다.
    # --------------------------------------------------------------------------

    revision = response.json()["revision"]


    # ==========================================================================
    # 10-2. Process Group 상태 변경에 사용할 데이터 생성
    # ==========================================================================

    payload = {

        # 방금 조회한 최신 Revision
        "revision": revision,

        # 상태를 변경할 Process Group ID
        "id": pg_id,

        # 변경할 상태
        #
        # RUNNING → 실행
        # STOPPED → 중지
        "state": state

    }


    # ==========================================================================
    # 10-3. 실제 Process Group 상태 변경 요청
    # ==========================================================================

    response = requests.put(

        # Process Group의 상태를 변경하는 API
        f"{NIFI_URL}/nifi-api/flow/process-groups/{pg_id}",

        headers=headers,

        # 위에서 만든 payload를 JSON 형태로 전달
        json=payload,

        verify=False

    )


    # API 요청 실패 시 예외 발생
    response.raise_for_status()


    # Airflow Task 로그에서 상태 변경 결과를 확인할 수 있도록 출력
    print(f"{state} : {pg_id}")

# 11. NiFi Process Group 작업 완료까지 대기
# ==============================================================================

def wait_until_finished(pg_id):
    """
    NiFi Process Group의 ETL 작업이 끝날 때까지 기다립니다.

    --------------------------------------------------------------------------
    왜 필요한가?
    --------------------------------------------------------------------------

    Airflow가 NiFi를 RUNNING으로 만든 직후 바로 다음 작업을 실행하면
    문제가 발생할 수 있습니다.

    예를 들어:

        Airflow
          ↓
        NiFi ODS 실행
          ↓
        아직 데이터 적재 중
          ↓
        Airflow가 DW 실행
          ↓
        ❌ ODS 데이터가 아직 완성되지 않음

    이런 상황을 방지하기 위해 NiFi의 실행 상태를 계속 확인합니다.

    --------------------------------------------------------------------------
    판단 기준
    --------------------------------------------------------------------------

    NiFi Process Group의

        activeThreadCount

    값을 확인합니다.

    activeThreadCount가

        0

    이 되면 현재 실행 중인 Thread가 없다는 의미이므로
    Process Group 작업이 종료된 것으로 판단합니다.
    """

    # --------------------------------------------------------------------------
    # NiFi API 인증 Token 발급
    # --------------------------------------------------------------------------

    token = get_nifi_token()


    # REST API 요청 Header
    headers = {

        "Authorization": f"Bearer {token}"

    }


    # ==========================================================================
    # Process Group 상태를 반복적으로 조회
    # ==========================================================================

    while True:

        # ----------------------------------------------------------------------
        # Process Group 상태 조회 API 호출
        # ----------------------------------------------------------------------

        response = requests.get(

            f"{NIFI_URL}/nifi-api/flow/process-groups/{pg_id}/status",

            headers=headers,

            verify=False

        )


        # HTTP 오류 발생 시 예외
        response.raise_for_status()


        # ----------------------------------------------------------------------
        # 현재 실행 중인 Thread 개수 확인
        # ----------------------------------------------------------------------
        #
        # NiFi 응답 JSON의 구조에서
        #
        # processGroupStatus
        #     ↓
        # aggregateSnapshot
        #     ↓
        # activeThreadCount
        #
        # 순서로 값을 찾아옵니다.
        # ----------------------------------------------------------------------

        active_thread = (

            response.json()

            ["processGroupStatus"]

            ["aggregateSnapshot"]

            ["activeThreadCount"]

        )


        # 현재 Thread 개수를 Airflow 로그에 출력
        print(f"Active Thread : {active_thread}")


        # ----------------------------------------------------------------------
        # 실행 중인 Thread가 0개인지 확인
        # ----------------------------------------------------------------------

        if active_thread == 0:

            # 더 이상 실행 중인 작업이 없으므로
            # NiFi Process Group 작업이 완료되었다고 판단합니다.
            print("Process Group Finished")

            # while True 반복문 종료
            break


        # ----------------------------------------------------------------------
        # 아직 작업이 진행 중이라면 3초 기다린 후 다시 상태를 확인합니다.
        #
        # 즉,
        #
        # 상태 확인
        #   ↓
        # 작업 중이면
        #   ↓
        # 3초 대기
        #   ↓
        # 다시 상태 확인
        #
        # 을 반복합니다.
        # ----------------------------------------------------------------------

        time.sleep(3)

# 12. ODS 영역
# ==============================================================================

def truncate_ods():
    """
    PostgreSQL ODS 테이블의 기존 데이터를 모두 삭제합니다.

    이번 ETL은 ODS 데이터를 새로 전체 적재하는 방식이므로
    NiFi ODS 작업을 실행하기 전에 기존 ODS 데이터를 TRUNCATE합니다.

    TRUNCATE는 테이블의 데이터를 전체 삭제하는 SQL입니다.

    주의:
        DELETE와 마찬가지로 데이터를 삭제하므로
        운영환경에서는 대상 테이블을 반드시 확인해야 합니다.
    """

    # --------------------------------------------------------------------------
    # Airflow Connection에 등록된 PostgreSQL 정보로 DB 연결
    # --------------------------------------------------------------------------

    hook = PostgresHook(

        postgres_conn_id=POSTGRES_CONN_ID

    )


    # --------------------------------------------------------------------------
    # ODS 테이블 초기화
    # --------------------------------------------------------------------------

    hook.run(
    """
    TRUNCATE TABLE
        ods_rental,
        ods_customer,
        ods_inventory,
        ods_film,
        ods_staff,
        ods_store,
        ods_address,
        ods_city,
        ods_country;
    """
    )

def start_ods():
    """
    NiFi ODS Process Group을 실행 상태(RUNNING)로 변경합니다.

    )

# 16. DW Autovacuum 설정
# ==============================================================================

def set_dw_autovacuum():
    """
    PostgreSQL DW 테이블의 Autovacuum 설정을 변경합니다.

    Autovacuum은 PostgreSQL이 테이블의 오래된 데이터를 정리하고
    통계 정보를 관리하는 기능입니다.

    여기서는 DW 테이블에 대해 Autovacuum 관련 설정을 적용합니다.
    """

    # PostgreSQL Connection 생성
    hook = PostgresHook(
        postgres_conn_id=POSTGRES_CONN_ID
    )


    # --------------------------------------------------------------------------
    # Autovacuum 설정을 적용할 DW 테이블 목록
    # --------------------------------------------------------------------------

    dw_tables = [
        "dw_rental",
        "dw_customer",
        "dw_inventory",
        "dw_film",
        "dw_staff",
        "dw_store",
        "dw_address",
        "dw_city",
        "dw_country"
    ]


    # --------------------------------------------------------------------------
    # 테이블 목록을 하나씩 순회하면서 설정 적용
    # --------------------------------------------------------------------------

    for table in dw_tables:

        sql = f"""
        ALTER TABLE {table} SET (
            autovacuum_enabled = true,
            autovacuum_vacuum_scale_factor = 0.01,
            autovacuum_analyze_scale_factor = 0.005
        );
        """

        # 해당 테이블에 SQL 실행
        hook.run(sql)

        # Airflow 로그에 어떤 테이블을 처리했는지 출력
        print(f"Autovacuum set for: {table}")

# 20. MART 생성
# ==============================================================================

def execute_mart():
    """
    PostgreSQL Stored Procedure를 호출하여 MART를 생성합니다.

    이 작업은 NiFi DW 작업이 완료된 이후 실행됩니다.

    즉,

        ODS 적재
          ↓
        DW 적재
          ↓
        Stored Procedure
          ↓
        MART 생성

    순서로 진행됩니다.
    """

    # --------------------------------------------------------------------------
    # PostgreSQL Connection 생성
    # --------------------------------------------------------------------------

    hook = PostgresHook(
        postgres_conn_id=POSTGRES_CONN_ID
    )


    # 현재 한국 시간 가져오기
    today = pendulum.now("Asia/Seoul")


    # --------------------------------------------------------------------------
    # 현재는 테스트를 위해 고정된 날짜를 사용합니다.
    #
    # 아래 값은 실제 ETL 대상 기간에 맞게 변경할 수 있습니다.
    #
    # 추후 자동 날짜 계산을 사용할 경우
    # 이 부분을 동적으로 변경할 수 있습니다.
    # --------------------------------------------------------------------------

    start_date = "2005-05-24"
    end_date = "2006-02-15"


    # --------------------------------------------------------------------------
    # PostgreSQL Stored Procedure 호출
    # --------------------------------------------------------------------------

    sql = f"""

        CALL sp_dm_rental_flag(

            '{start_date}',

            '{end_date}'

        );

    """


    # Stored Procedure 실행
    hook.run(sql)


    # Airflow 로그에 실행 결과 출력
    print(
        f"""

        MART LOAD COMPLETE

        START DATE : {start_date}

        END DATE   : {end_date}

        """
    )

# 21. DAG 정의
# ==============================================================================

# with DAG(...) as dag:
#
# 아래 블록 안에서 작성한 PythonOperator들이
# 실제 Airflow Task로 생성됩니다.
#
# DAG는 쉽게 말하면
#
# "이 작업들을 어떤 순서로 실행할 것인가?"
#
# 를 정의하는 영역입니다.
# ==============================================================================

with DAG(

    # --------------------------------------------------------------------------
    # Airflow UI에 표시되는 DAG의 고유 이름
    # --------------------------------------------------------------------------

    dag_id="RENTAL_ETL",


    # --------------------------------------------------------------------------
    # DAG의 시작 기준일
    #
    # 2026년 1월 1일 이후의 스케줄을 기준으로 DAG를 관리합니다.
    # --------------------------------------------------------------------------

    start_date=pendulum.datetime(

        2026,

        1,

        1,

        tz="Asia/Seoul"

    ),


    # --------------------------------------------------------------------------
    # DAG 실행 스케줄
    #
    # "0 1 * * *"
    #
    # = 매일 새벽 1시 실행
    #
    # Cron 형식:
    #
    # ┌──────── 분
    # │ ┌────── 시
    # │ │ ┌──── 일
    # │ │ │ ┌── 월
    # │ │ │ │ ┌ 요일
    # │ │ │ │ │
    # 0 1 * * *
    #
    # --------------------------------------------------------------------------

    schedule="0 1 * * *",


    # --------------------------------------------------------------------------
    # 과거 날짜의 DAG Run을 자동으로 생성할지 여부
    #
    # False:
    #     DAG를 처음 활성화했을 때 과거 실행분을 자동으로 만들지 않습니다.
    #
    # --------------------------------------------------------------------------

    catchup=False,


    # --------------------------------------------------------------------------
    # Airflow UI에서 DAG를 찾기 쉽게 하기 위한 태그
    # --------------------------------------------------------------------------

    tags=[

        "NiFi",

        "ETL",

        "Rental"

    ]

) as dag:


    # ==========================================================================
    # 22. ODS TRUNCATE Task
    # ==========================================================================

    # 위에서 만든 truncate_ods() Python 함수를
    # Airflow Task로 등록합니다.

    truncate = PythonOperator(

        # Airflow UI에 표시되는 Task ID
        task_id="truncate_ods",

        # 실제로 실행할 Python 함수
        python_callable=truncate_ods

    )


    # ==========================================================================
    # 23. ODS NiFi 실행 Task
    # ==========================================================================

    ods_start = PythonOperator(

        task_id="start_ods",

        # start_ods() 함수 실행
        python_callable=start_ods

    )


    # ==========================================================================
    # 24. ODS 완료 대기 Task
    # ==========================================================================

    ods_wait = PythonOperator(

        task_id="wait_ods",

        # NiFi ODS 작업이 끝날 때까지 기다리는 함수
        python_callable=wait_ods

    )


    # ==========================================================================
    # 25. ODS NiFi 중지 Task
    # ==========================================================================

    ods_stop = PythonOperator(

        task_id="stop_ods",

        # NiFi ODS Process Group을 STOPPED 상태로 변경
        python_callable=stop_ods

    )


    # ==========================================================================
    # 26. DW Autovacuum 설정 Task
    # ==========================================================================

    dw_autovacuum = PythonOperator(

        task_id="set_dw_autovacuum",

        # PostgreSQL DW 테이블의 Autovacuum 설정
        python_callable=set_dw_autovacuum

    )


    # ==========================================================================
    # 27. DW NiFi 실행 Task
    # ==========================================================================

    dw_start = PythonOperator(

        task_id="start_dw",

        # NiFi DW Process Group 실행
        python_callable=start_dw

    )


    # ==========================================================================
    # 28. DW 완료 대기 Task
    # ==========================================================================

    dw_wait = PythonOperator(

        task_id="wait_dw",

        # NiFi DW 작업이 끝날 때까지 기다림
        python_callable=wait_dw

    )


    # ==========================================================================
    # 29. DW NiFi 중지 Task
    # ==========================================================================

    dw_stop = PythonOperator(

        task_id="stop_dw",

        # NiFi DW Process Group 중지
        python_callable=stop_dw

    )


    # ==========================================================================
    # 30. MART 생성 Task
    # ==========================================================================

    mart = PythonOperator(

        task_id="execute_mart",

        # PostgreSQL Stored Procedure 실행
        python_callable=execute_mart

    )


    # ==========================================================================
    # 31. Task 실행 순서 정의
    # ==========================================================================

    # ">>"는 앞의 Task가 성공적으로 끝난 다음
    # 뒤의 Task를 실행하라는 의미입니다.
    #
    # 따라서 아래 코드는 다음과 같은 순서를 의미합니다.
    #
    # truncate
    #     ↓
    # ods_start
    #     ↓
    # ods_wait
    #     ↓
    # ods_stop
    #     ↓
    # dw_autovacuum
    #     ↓
    # dw_start
    #     ↓
    # dw_wait
    #     ↓
    # dw_stop
    #     ↓
    # mart
    #
    # 즉, 위쪽 Task가 성공해야 다음 Task가 실행됩니다.
    # ==========================================================================

    (

        truncate

        >> ods_start

        >> ods_wait

        >> ods_stop

        >> dw_autovacuum

        >> dw_start

        >> dw_wait

        >> dw_stop

        >> mart

    )
